cmake_minimum_required(VERSION 3.20)
project(latticeforge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(latticeforge_core src/model.cpp src/search.cpp)
target_include_directories(latticeforge_core PUBLIC include)
add_executable(latticeforge-search src/main.cpp)
target_link_libraries(latticeforge-search PRIVATE latticeforge_core)
add_executable(latticeforge-tests tests/test_engine.cpp)
target_link_libraries(latticeforge-tests PRIVATE latticeforge_core)
enable_testing()
add_test(NAME latticeforge-tests COMMAND latticeforge-tests)
